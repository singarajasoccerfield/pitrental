Integrasi Google Spreadsheet

Belum dikonfigurasi

Panduan Singkat Pemasangan (1 Menit)
  1. Buka dokumen Google Spreadsheet baru milik Anda.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus semua kode lama, lalu tempel kode skrip di bawah ini. Skrip ini akan otomatis membuat tab per unit alat dan menyimpan foto ke folder Google Drive "DOKUMENTASI_HM_PIT_RENTAL".
  4. Klik Deploy > New deployment (atau Manage deployments > Edit > Version: New version jika memperbarui).
  5. Ubah Who has access menjadi Anyone (Siapa saja), lalu klik Deploy.
  6. Salin URL Web App yang dihasilkan dan tempel pada input di atas.
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(30000);
  
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = {};
    
    // 1. Ekstrak data JSON payload atau query parameters
    if (e && e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (errJson) {
        data = e.parameter || {};
      }
    } else if (e && e.parameter) {
      data = e.parameter;
    }

    // 2. Handler Khusus: Upload Foto Langsung ke Google Drive
    if (data.action === "uploadImage" || (data.imageBlobBase64 && data.isPhotoUploadOnly)) {
      var folderName = data.folderName || "DOKUMENTASI_HM_PIT_RENTAL";
      var folders = DriveApp.getFoldersByName(folderName);
      var targetFolder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

      var contentType = data.imageMimeType || "image/jpeg";
      var rawBase64 = data.imageBlobBase64.replace(/^data:image\/\w+;base64,/, "");
      var decodedBytes = Utilities.base64Decode(rawBase64);
      var fileName = (data.fileName || "FOTO_HM_" + new Date().getTime()) + ".jpg";
      var fileBlob = Utilities.newBlob(decodedBytes, contentType, fileName);

      var driveFile = targetFolder.createFile(fileBlob);
      driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

      var fileUrl = "https://drive.google.com/uc?export=view&id=" + driveFile.getId();

      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        fileUrl: fileUrl,
        fileId: driveFile.getId(),
        message: "Foto berhasil diunggah ke Google Drive!"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    // 3. Ambil nama unit alat lengkap untuk dijadikan nama Sheet/Tab
    var rawSheetName = (e && e.parameter && e.parameter.sheetTargetName) 
                       || data.sheetTargetName 
                       || data.equipmentCode 
                       || "UNIT_ALAT";
    
    var cleanSheetName = String(rawSheetName)
      .replace(/[\[\]\*?\/\:\\]/g, "-")
      .replace(/\s+/g, " ")
      .trim();
      
    if (!cleanSheetName) {
      cleanSheetName = "ALAT_BERAT";
    }
    if (cleanSheetName.length > 90) {
      cleanSheetName = cleanSheetName.substring(0, 90).trim();
    }
    
    // 4. Cari sheet dengan nama unit alat tersebut, atau buat baru jika belum ada
    var sheet = ss.getSheetByName(cleanSheetName);
    if (!sheet) {
      sheet = ss.insertSheet(cleanSheetName);
    }
    
    // 5. Buat baris header jika tab ini baru dibuat / masih kosong
    if (sheet.getLastRow() === 0) {
      var headers = [
        "ID Log", 
        "Tanggal", 
        "Nama Unit Alat",
        "Nama Operator", 
        "Lokasi / Site",
        "Jam Mulai", 
        "Jam Selesai", 
        "Jam Bucket", 
        "Jam Breaker", 
        "Total Jam Kerja", 
        "HM Awal", 
        "HM Akhir", 
        "Total HM", 
        "Catatan Pekerjaan", 
        "Link Foto Google Drive",
        "Waktu Sinkronisasi"
      ];
      sheet.appendRow(headers);
      
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setFontWeight("bold");
      headerRange.setBackground("#f59e0b");
      headerRange.setFontColor("#0f172a");
      headerRange.setHorizontalAlignment("center");
      sheet.setFrozenRows(1);
    }

    // 6. Jika ada foto disertakan bersama log dan belum diupload terpisah
    var finalPhotoUrl = data.photoUrl || "";
    if (!finalPhotoUrl && data.imageData && data.imageData.startsWith("data:image")) {
      try {
        var folderName2 = "DOKUMENTASI_HM_PIT_RENTAL";
        var folders2 = DriveApp.getFoldersByName(folderName2);
        var f2 = folders2.hasNext() ? folders2.next() : DriveApp.createFolder(folderName2);
        var bytes = Utilities.base64Decode(data.imageData.replace(/^data:image\/\w+;base64,/, ""));
        var fBlob = Utilities.newBlob(bytes, "image/jpeg", "HM_" + (data.equipmentCode||"UNIT") + "_" + new Date().getTime() + ".jpg");
        var upFile = f2.createFile(fBlob);
        upFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        finalPhotoUrl = "https://drive.google.com/uc?export=view&id=" + upFile.getId();
      } catch (errPhoto) {}
    }
    
    // 7. Masukkan baris log
    var unitFullName = data.equipmentCode || cleanSheetName;
    var operatorName = data.operator || "-";
    
    sheet.appendRow([
      data.id || ("LOG-" + new Date().getTime()),
      data.date || "",
      unitFullName,
      operatorName,
      data.location || "-",
      data.startTime || "",
      data.endTime || "",
      Number(data.bucketHours) || 0,
      Number(data.breakerHours) || 0,
      Number(data.totalHours) || 0,
      Number(data.hmAwal) || 0,
      Number(data.hmAkhir) || 0,
      Number(data.totalHm) || 0,
      data.notes || "",
      finalPhotoUrl || "-",
      new Date().toLocaleString("id-ID", { timeZone: "Asia/Jakarta" })
    ]);
    
    for (var col = 1; col <= 16; col++) {
      sheet.autoResizeColumn(col);
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      sheetName: cleanSheetName,
      photoUrl: finalPhotoUrl,
      message: "Sukses tersimpan di sheet tab: " + cleanSheetName
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
