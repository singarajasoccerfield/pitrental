function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "getConfig";
  if (action === "getConfig") {
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      config: getSecretAppConfig()
    })).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "online" })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    if (data.action === "getConfig") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        config: getSecretAppConfig()
      })).setMimeType(ContentService.MimeType.JSON);
    }
    var sheetName = String(data.sheetTargetName || data.equipmentCode || "UNIT_ALAT").replace(/[\[\]\*?\/\:\\]/g, "-").trim();
    var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID Log","Tanggal","Nama Unit Alat","Nama Operator","Lokasi / Site","Jam Mulai","Jam Selesai","Jam Bucket","Jam Breaker","Total Jam Kerja","HM Awal","HM Akhir","Total HM","Catatan Pekerjaan","Link Foto Google Drive","Waktu Sinkronisasi"]);
    }
    var rows = sheet.getDataRange().getValues();
    var foundRow = -1;
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(data.id)) { foundRow = i + 1; break; }
    }
    if (data.action === "deleteLog") {
      if (foundRow > 1) sheet.deleteRow(foundRow);
      return ContentService.createTextOutput(JSON.stringify({ status: "deleted" })).setMimeType(ContentService.MimeType.JSON);
    }
    var vals = [data.id, data.date, data.equipmentCode, data.operator, data.location, data.startTime, data.endTime, Number(data.bucketHours)||0, Number(data.breakerHours)||0, Number(data.totalHours)||0, Number(data.hmAwal)||0, Number(data.hmAkhir)||0, Number(data.totalHm)||0, data.notes||"", data.photoDriveUrl||"-", new Date().toLocaleString("id-ID")];
    if (foundRow > 1) sheet.getRange(foundRow, 1, 1, vals.length).setValues([vals]);
    else sheet.appendRow(vals);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } finally { lock.releaseLock(); }
}
